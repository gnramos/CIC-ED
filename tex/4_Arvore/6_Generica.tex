\begin{frame}[fragile,label=notas]
	\framesubtitle{Aplicação: Árvore de Busca (de Soluções)}
	
	\begin{itemize}
		\item Game Trees são árvores que representam as  possibilidades de jogadas para um jogador a partir de um estado do jogo.\pause
		\item Os filhos de um nó representam todas as possibilidades a partir daquela situação do jogo.\pause
		\item Função \emph{avalia} retorna um valor representando o quão bom um estado do jogo (posição do tabuleiro).\pause
	\end{itemize}
	\begin{columns}[T]
		\column{.4\textwidth}\centering
			\begin{lstlisting}
				typedef struct gt_node_t {
     int info;
     struct gt_node_t *pai;
     struct gt_node_t *filho;
     struct gt_node_t *irmao;
} gt_node_t;
			\end{lstlisting}

\begin{frame}[label=notas]
	\framesubtitle{Aplicação: Árvore de Busca (de Soluções) - Jogo da Velha}
	
	\begin{itemize}
		\item Altura da árvore indica o número de jogadas adiante que se deseja prever.
		\item A decisão deve maximizar a função \emph{avalia} para o jogador (e minimizar  para o adversário).
			\subitem{A função de avaliação retorna o valor do número de linhas colunas e diagonais abertas para o programa menos o número de linhas, colunas e diagonais abertas ao adversário.}
	\end{itemize}

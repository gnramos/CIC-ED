\section{Ponteiros}
\begin{frame}[label=notas]
	\framesubtitle{\subsecname\ (apontador)}

	\begin{beamerboxesrounded}{}
		Tipo de dado de uma linguagem de programação. O valor armazenado se refere \emph{diretamente} a um outro valor alocado em outra área da memória - é o endereço deste.
	\end{beamerboxesrounded}
	\begin{figure}
		\centering
		\begin{tikzpicture}[node distance=1em,minimum width=5.5em, minimum height=1.5em]
			\node[block,draw=redUnB,label=above:$ptr\_int$]	(ptr) 	{1048736};
			\node[block,right=of ptr,label=above:$int$]    	(int) 	{2010};
			\node[block,below=of int,label=below:$int$]    	(int1)	{2011};
			\node[block,right=of int,label=above:$char$]   	(int2)	{CIC};
			\node[block,below=of int2,label=below:$char$]  	(int3)	{UnB};
			\node[block,right=of int2,label=above:$int$]   	(int4)	{2014};
			\node[block,below=of int4,label=below:$char$]  	(int5)	{ED};
			\draw[->,line] (ptr) |- (int1);
		\end{tikzpicture}
	\end{figure}

	\end{alertblock}
\end{frame}

\begin{frame}[label=main]
	\begin{exampleblock}{Notação Assintótica (Grande-O/Big-O Notation)}
		Notação matemática usada para analisar o comportamento  de funções
		\subitem{utilizada para descrever o uso de recursos computacionais}
	\end{exampleblock}
	\vfill

	\begin{itemize}
		\item prever o comportamento do algoritmo
		\item determinar qual algoritmo utilizar
	\end{itemize}

\begin{frame}[label=main]
	\begin{exampleblock}{Dominância Assintótica - $O(g(n)) = f(n)$}
		Uma função $g(n)$ domina assintoticamente outra função $f(n)$ se existem 
		duas constantes positivas $c$ e $n_0$ tais que:
		$0 \leq f(n) \leq c{\cdot}g(n), ^{\forall}n \geq n_0$.
	\end{exampleblock}
	\vfill
	\begin{center}%
		\includegraphics[width=.4\textwidth]{Introduction_BigOGraph}%
	\end{center}
\end{frame}

\begin{frame}[label=graduacao]
	Operações:\\\vfill
	\begin{center}
		\begin{tabular}{rl}
			$f(n)$ 					& $\rightarrow$  $O(f(n))$\\
			$c{\cdot}f(n)$ 			& $\rightarrow$  $O(f(n))$ \textcolor{gray}{c constante}\\
			$O(f(n)) + O(f(n))$ 	& $\rightarrow$  $O(f(n))$\\
			$O(O(f(n)))$ 			& $\rightarrow$  $O(f(n))$\\
			$O(f(n) + g(n))$ 		& $\rightarrow$  $O(max(f(n),g(n)))$\\
			$O(f(n)){\cdot}O(g(n))$ & $\rightarrow$  $O(f(n){\cdot}g(n))$\\
			$f(n){\cdot}O(g(n))$ 	& $\rightarrow$  $O(f(n){\cdot})g(n))$\\
		\end{tabular}
	\end{center}
\end{frame}

\begin{frame}[fragile,label=main]
	Exemplo: Comparar cada elemento de um vetor aos outros elementos.
	\subitem{função de custo: \alert<1>{comparação}}
	\begin{columns}[T]
		\column{.35\textwidth}
			\begin{lstlisting}[title=Algoritmo,morekeywords={compare}]
				para i /in [1, n]
				    para j /in [1, n]
				        se i != j
				            compare(i,j)
			\end{lstlisting}
			\pause
			$g(n) = n(n-1)$
		\column{.35\textwidth}
			\begin{lstlisting}[title= Algoritmo Otimizado,morekeywords={compare}]
				para i /in [1, n)
				    para j /in [i+1,n]
				        compare(i,j)
			\end{lstlisting}
			\pause
			$$g(n) = \sum_{i=1}^{n-1}(n-i) = \frac{n(n-1)}{2}$$
	\end{columns}
	\vfill
	E as complexidades?	
\end{frame}

\begin{frame}[label=main]
	Casos de estudo da função de custo:
	\begin{description}
		\item[Melhor:] definida pelo menor número de passos executados 
		para qualquer instância de tamanho $n$
		\item[Médio:] definida pela média do número de passos executados 
		para qualquer instância de tamanho $n$
		\item[Pior:] definida pelo maior número de passos executados para 
		qualquer instância de tamanho $n$
	\end{description}
	\pause\vfill
	\begin{block}{Complexidade Assintótica}
		\begin{itemize}
			\item Para pequenos valores de $n$, a maioria dos algoritmos não representa problemas.
			\item Estuda-se a complexidade somente para \alert<2>{grandes valores de $n$}.
			\item O comportamento assintótico de $g(n$) representa o limite do 
			comportamento do custo quando $n$ cresce.
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}[label=main]
	Técnicas de análise de algoritmos
		\begin{itemize}
			\item considerar memória infinita
			\item não considerar o sistema operacional nem o compilador
			\item analisar o algoritmo e não o programa
			\item levar em conta o tamanho das entradas
			\item ter cuidado ao escolher a função de custo:
				\begin{itemize}
					\item atribuição
					\item adição
					\item multiplicação
					\item comparação
					\item etc.
				\end{itemize}
		\end{itemize}
\end{frame}

\begin{frame}[label=main]
	\framesubtitle{Dicas}
		\begin{itemize}
			\item A complexidade de um laço é igual ao número de comandos internos 
			vezes o número de vezes que ele é executado
			\vfill
			\item A complexidade de laços aninhados é o produto dos tamanhos dos laços
			\vfill
			\item Para uma sequência de laços do algoritmo, a complexidade é a 
			do laço de maior custo
			\vfill
			\item Em testes condicionais, a complexidade é a maior das duas 
			partes (if/else) do teste
		\end{itemize}
\end{frame}

\section{Eficiência}
\begin{frame}[label=main]
	Para dois algoritmos que executem a mesma tarefa, a diferença entre 
	complexidades pode ser muito mais significante que a entre hardware/software.
	\\\vfill\pause
	Supondo:
	\begin{description}
		\item[Insertion Sort] com custo $c_1n^2$, rodando no computador $A$
		\item[Merge Sort] com custo $c_2\cdot{n}\log(n)$, rodando no computador $B$
	\end{description}
	\vfill\pause
	Se $n = 10^7$ e:
	\begin{description}
		\item[$A$] executa $10^{10}$ instruções por segundo
		\item[$B$] executa $10^{7}$ instruções por segundo
		\item[$c_1 = 2$] devido ao superprogramador
		\item[$c_2 = 50$] devido a preguiça do estagiário
		% g_A = 2*(10^{10})^{2} = 2*(10ˆ{20})
		% g_B = 50*(10^{7})*log(10^{7}) = 3.5*10ˆ9
	\end{description}
\end{frame}

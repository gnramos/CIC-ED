\subsection*{Exercício}
\begin{frame}[label=main]
	\begin{block}{Entrada}
		Um arquivo com, no máximo, $n$ inteiros positivos distintos, cada um de valor $\leq 10^7$. Não há qualquer outro dado relacionado ao número.
	\end{block}
	\vfill\pause
	\begin{exampleblock}{Saída}
		Uma lista com os inteiros (de entrada) \emph{em ordem crescente}.
	\end{exampleblock}
	\vfill\pause
	\begin{alertblock}{Restrições}
		Em torno de 1Mb de memória RAM, espaço a vontade no disco rígido. Processamento em poucos minutos.
	\end{alertblock}
\end{frame}
